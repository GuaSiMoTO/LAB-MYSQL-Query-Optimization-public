-- Warm-Up Queries
-- These simple queries will help you get familiar with the tables and data:
use Final_Attack_Imperial;

-- Calculate the average power of the weapons assigned to each Imperial team.
SELECT ImperialTeams.name, Weapons.name, AVG(Weapons.power) AS avg_power 
FROM ImperialTeams
INNER JOIN ImperialTeamWeapons 
ON ImperialTeams.team_id =  ImperialTeamWeapons.imperial_team_id
INNER JOIN Weapons
ON Weapons.weapon_id = ImperialTeamWeapons.weapon_id
GROUP BY ImperialTeams.name, Weapons.name;

-- Get the total strength of the Rebel teams stationed at each base.
SELECT Bases.base_id, SUM(RebelTeams.total_force) AS total_strength
FROM RebelTeams
JOIN Bases
ON Bases.base_id = RebelTeams.base_id
GROUP BY Bases.base_id;

-- Views
-- These exercises focus on creating views to simplify complex queries and make them reusable.

-- 1. Rebel Team with the Highest and Lowest Total Strength:
-- Create a view that identifies the Rebel team with the highest and lowest total strength.

CREATE VIEW highest_and_lowest_strength_VIEW AS
(SELECT name, total_force as strength
FROM RebelTeams 
WHERE total_force = (SELECT MAX(total_force) FROM RebelTeams) limit 1)
UNION
(SELECT name, total_force AS strength 
FROM RebelTeams 
WHERE total_force = (SELECT MIN(total_force) FROM RebelTeams) limit 1);

-- 2. Summary of Resources by Location
-- Create a view to get a summary of the resources available at each location, grouped by location.

CREATE VIEW summary_resources_VIEW AS
SELECT Locations.name, GROUP_CONCAT(COALESCE(Bases.resources, '')) AS summary_resources FROM Locations
JOIN Bases
On Bases.base_id = Locations.location_id
GROUP BY Locations.name;


-- 3. Imperial Teams and Weapon Count
-- Create a view to list all Imperial teams and the total number of weapons assigned to each team, ordered by the total number of weapons.

CREATE VIEW total_weapons_Imperial_teams_VIEW AS
SELECT ImperialTeams.name, COUNT(Weapons.name) AS number_of_weapons FROM ImperialTeams
INNER JOIN ImperialTeamWeapons
ON ImperialTeams.team_id = ImperialTeamWeapons.imperial_team_id
INNER JOIN Weapons
ON ImperialTeamWeapons.weapon_id = Weapons.weapon_id
GROUP BY ImperialTeams.name
ORDER BY number_of_weapons;

-- SUBQUERIES

-- Subquery: Imperial Team with the Highest Average Weapon Power
-- Write a subquery that returns the name of the Imperial team with the highest average power of assigned weapons. Include the average power.

SELECT ImperialTeams.name, (
    SELECT AVG(Weapons.power)
    FROM Weapons
    WHERE Weapons.weapon_id IN (SELECT ImperialTeamWeapons.weapon_id FROM ImperialTeamWeapons WHERE ImperialTeams.team_id = ImperialTeamWeapons.imperial_team_id)) AS avg_power
FROM ImperialTeams
ORDER BY avg_power DESC
LIMIT 1;

-- CREATE AND USE VIEWS
– 1.  Weapons Used in Successful Attacks

– Step 1: Create a view to calculate the average success probability of attacks
.
CREATE VIEW view_avg_attacks AS
SELECT AVG(success_probability) AS AVG_probability_attacks FROM Attacks;

– Step 2: Use the view to filter weapons assigned to teams in successful attacks.

SELECT ImperialTeams.name, Weapons.name FROM  ImperialTeams
INNER JOIN ImperialTeamWeapons
ON ImperialTeamWeapons.imperial_team_id =  ImperialTeams.team_id
INNER JOIN Weapons
ON ImperialTeamWeapons.weapon_id = Weapons.weapon_id
WHERE Weapons.power > (SELECT * FROM view_avg_attacks);

-- 2. Rebel Teams in Bases with the Lowest Security Level

–  Step 1: Create a view to identify bases with the lowest security level
.
CREATE VIEW view_lowest_security_level AS
SELECT base_id FROM Bases
WHERE security_level = 'Low';
– Step 2: Use the view to find Rebel teams assigned to these bases.

SELECT * FROM RebelTeams
WHERE RebelTeams.base_id 
IN (SELECT base_id FROM view_lowest_security_level);
